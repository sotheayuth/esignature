Package — Flutter

class SignCard extends StatelessWidget {
  const SignCard({super.key, required this.width,
      required this.title, required this.name, this.ink});
  final double width;                 // the ONLY absolute number
  final String title, name;
  final ImageProvider? ink;

  @override
  Widget build(BuildContext context) => SizedBox(
    width: width,
    child: AspectRatio(
      aspectRatio: 2,                            // 94 x 47
      child: LayoutBuilder(builder: (_, box) {
        final u = box.maxWidth / 100;            // u = 1% of the card
        Widget row(String s, TextStyle st) => Text(s, maxLines: 1,
            softWrap: false, overflow: TextOverflow.ellipsis,
            textAlign: TextAlign.center, style: st);
        TextStyle f(double s, double h) =>
            TextStyle(fontFamily: 'Moul', fontSize: s * u, height: h);
        return Stack(children: [
          Positioned(top: 2.5 * u, left: 2 * u, right: 2 * u,
              child: row(title, f(4.4, 1.7))),     // title — top
          Positioned(bottom: 2 * u, left: 2 * u, right: 2 * u,
              child: row(name, f(4.4, 1.8))),      // name — bottom
          if (ink != null) Positioned.fill(        // sign — middle
              left: 15 * u, right: 15 * u, top: 12 * u, bottom: 12 * u,
              child: Image(image: ink!,
                  fit: BoxFit.contain)),           // never stretch
        ]);
      }),
    ),
  );
}
